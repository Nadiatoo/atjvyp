// ========== 条件1：均线多头排列 ==========
MA5:=MA(CLOSE,5);
MA8:=MA(CLOSE,8);
MA13:=MA(CLOSE,13);
COND1:=MA5 > MA8 AND MA8 > MA13;

// ========== 条件2：成交量倍量放量 ==========
VOL_DBL1:=VOL > REF(VOL,1) * 2;
VOL_DBL2:=VOL > REF(VOL,2) * 2;
DBL_VOL_DAY:=VOL_DBL1 OR VOL_DBL2;
DBL_DAY_CNT:=COUNT(DBL_VOL_DAY,1);
DBL_DAY_CNT:=DBL_DAY_CNT+COUNT(REF(DBL_VOL_DAY,1),1);
DBL_DAY_CNT:=DBL_DAY_CNT+COUNT(REF(DBL_VOL_DAY,2),1);
DBL_DAY_CNT:=DBL_DAY_CNT+COUNT(REF(DBL_VOL_DAY,3),1);
DBL_DAY_CNT:=DBL_DAY_CNT+COUNT(REF(DBL_VOL_DAY,4),1);
COND2:=DBL_DAY_CNT >= 3;

// ========== 条件3：主力控盘连续3日>10 ==========
ZJTJ_VAL:=ZJTJ(CLOSE,VOL);
ZJTJ_C1:=ZJTJ_VAL > 10;
ZJTJ_C2:=REF(ZJTJ_VAL,1) > 10;
ZJTJ_C3:=REF(ZJTJ_VAL,2) > 10;
COND3:=ZJTJ_C1 AND ZJTJ_C2 AND ZJTJ_C3;

// ========== 条件4：近10日出现过涨停 ==========
LIMIT_UP:=CLOSE/REF(CLOSE,1) >= 1.099;
HAS_LIMIT:=COUNT(LIMIT_UP,1);
HAS_LIMIT:=HAS_LIMIT+COUNT(REF(LIMIT_UP,1),1);
HAS_LIMIT:=HAS_LIMIT+COUNT(REF(LIMIT_UP,2),1);
HAS_LIMIT:=HAS_LIMIT+COUNT(REF(LIMIT_UP,3),1);
HAS_LIMIT:=HAS_LIMIT+COUNT(REF(LIMIT_UP,4),1);
HAS_LIMIT:=HAS_LIMIT+COUNT(REF(LIMIT_UP,5),1);
HAS_LIMIT:=HAS_LIMIT+COUNT(REF(LIMIT_UP,6),1);
HAS_LIMIT:=HAS_LIMIT+COUNT(REF(LIMIT_UP,7),1);
HAS_LIMIT:=HAS_LIMIT+COUNT(REF(LIMIT_UP,8),1);
HAS_LIMIT:=HAS_LIMIT+COUNT(REF(LIMIT_UP,9),1);
COND4:=HAS_LIMIT >= 1;

// ========== 综合选股条件 ==========
SELECTION:=COND1 AND COND2 AND COND3 AND COND4;
SELECTION;
